# StarHermit engine script game: one package, played natively by the
# StarHermit client and in browsers by the engine's web runtime. Browsers open
# launch=index.html, the loader `npm run package` writes (it pins the engine
# on wiki.starhermit.com; never edit it). Re-run `npm run package` after every
# change and commit index.html + starhermit-files.json. server.js is the one
# platform script: it referees hosted matches (room-bound sessions) and posts
# solo match scores to the leaderboard.
launch=index.html
engine.api=1
engine.entry=native/main.ts
engine.version=1.24.0
engine.minor=19
engine.capabilities=platform,storage,clipboard

# Keyboard controls (players rebind keys and pads in the runtime's Options).
control.up=ArrowUp | Move mallet up
control.down=ArrowDown | Move mallet down
control.left=ArrowLeft | Move mallet left
control.right=ArrowRight | Move mallet right
control.pause=Escape | Pause / match menu
control.undo=KeyZ | Undo
control.camera=KeyC | Recentre camera
control.hint=KeyH | Hint
control.board=KeyB | Board state (described)
engine.group.up=Mallet
engine.group.pause=Match

# Gamepad defaults for the same actions (the left stick moves, Start pauses).
engine.gamepad.up=LeftStickUp+DpadUp
engine.gamepad.down=LeftStickDown+DpadDown
engine.gamepad.left=LeftStickLeft+DpadLeft
engine.gamepad.right=LeftStickRight+DpadRight
engine.gamepad.pause=Start
engine.gamepad.undo=West
engine.gamepad.camera=RightStickPress
engine.gamepad.hint=North
engine.gamepad.board=Select
